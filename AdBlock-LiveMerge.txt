  }

  function jqParts(text, separator) {
    const result = []; let start = 0, quoted = false, escaped = false;
    const stack = [], pairs = { ")":"(", "]":"[", "}":"{" };
    for (let i = 0; i < text.length; i++) {
      const c = text[i];
      if (quoted) { if (escaped) escaped = false; else if (c === "\\") escaped = true; else if (c === '"') quoted = false; continue; }
      if (c === '"') { quoted = true; continue; }
      if ("([{".includes(c)) { stack.push(c); continue; }
      if (")]}".includes(c)) { if (stack.pop() !== pairs[c]) fail("FleaMarket jq grouping changed"); continue; }
      const word = separator !== "|";
      if (!stack.length && text.startsWith(separator, i) &&
          (word ? !/[A-Za-z0-9_]/.test(text[i - 1] || "") && !/[A-Za-z0-9_]/.test(text[i + separator.length] || "") : text[i + 1] !== "=")) {
        result.push(text.slice(start, i).trim()); i += separator.length - 1; start = i + 1;
      }
    }
    if (quoted || stack.length) fail("FleaMarket jq syntax changed");
    result.push(text.slice(start).trim()); return result;
  }
  function jqCode(text) {
    let quoted = false, escaped = false;
    for (let i = 0; i < text.length; i++) {
      const c = text[i];
      if (quoted) { if (escaped) escaped = false; else if (c === "\\") escaped = true; else if (c === '"') quoted = false; }
      else if (c === '"') quoted = true;
      else if (c === "#") return text.slice(0, i).trim();
    }
    return text.trim();
  }
  function jqUngroup(text) {
    let value = text.trim();
    while (value.startsWith("(") && value.endsWith(")")) {
      let depth = 0, quoted = false, escaped = false, outer = true;
      for (let i = 0; i < value.length; i++) {
        const c = value[i];
        if (quoted) { if (escaped) escaped = false; else if (c === "\\") escaped = true; else if (c === '"') quoted = false; continue; }
        if (c === '"') quoted = true;
        else if (c === "(") depth++;
        else if (c === ")" && --depth === 0 && i !== value.length - 1) { outer = false; break; }
      }
      if (!outer) break;
      value = value.slice(1, -1).trim();
    }
    return value;
  }
  function jqRootMutation(clause) {
    const mutation = /^\.data\.[A-Za-z_][A-Za-z0-9_]*(?:\[\]\??)?(?:\.[A-Za-z_][A-Za-z0-9_]*(?:\[\]\??)?)*\s*\|=\s*((?:map|del)\([\s\S]*\))$/.exec(clause);
    if (!mutation) return false;
    const grouped = mutation[1].slice(mutation[1].indexOf("("));
    return jqUngroup(grouped) !== grouped.trim();
  }
  function jqComparisons(text, alias) {

    if (jqParts(text, "or").length !== 1) return null;
    const parts = jqParts(text, "and");

      return values.some(x => x === null) ? null : values.flat();
    }

    if (!match) return null;
