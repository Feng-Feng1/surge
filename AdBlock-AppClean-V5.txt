(function () {
  "use strict";
  var label = typeof $argument === "string" ? $argument.split("&")[0] : "";
  var debug = typeof $argument === "string" && /(?:^|&)debug=1(?:&|$)/.test($argument);
  var changed = false;
  var edits = 0;
  var own = Object.prototype.hasOwnProperty;
  function object(x) { return x !== null && typeof x === "object" && !Array.isArray(x); }
  function has(x, k) { return object(x) && own.call(x, k); }
  function active(x) {
    if (Array.isArray(x)) return x.length > 0;
    if (object(x)) return Object.keys(x).length > 0;
    return x !== undefined && x !== null && x !== false && x !== 0 && x !== "" && x !== "0" && x !== "false";
  }
  function flag(x) { return x === true || x === 1 || x === "1" || x === "true"; }
  function mark() { changed = true; edits++; }
  function get(root, path) {
    var parts = path.split(".");
    for (var i = 0; i < parts.length; i++) {
      if (root === null || typeof root !== "object") return undefined;
      root = root[parts[i]];
    }
    return root;
  }
  function setExisting(root, path, value) {
    var parts = path.split(".");
    var key = parts.pop();
    var parent = parts.length ? get(root, parts.join(".")) : root;
    if (!has(parent, key)) return;
    var current = parent[key];
    // 服务端改了字段类型时不猜测新结构；数字开关同时保留原数字／字符串类型。
    if (Array.isArray(value) && !Array.isArray(current)) return;
    if (object(value) && !object(current)) return;
    if (value === 0 || value === "0") {
      if (typeof current === "number" && isFinite(current)) value = 0;
      else if (typeof current === "string" && /^-?\d+(?:\.\d+)?$/.test(current)) value = "0";
      else return;
    } else if (typeof value !== typeof current) return;
    if (JSON.stringify(current) !== JSON.stringify(value)) {
      parent[key] = value; mark();
    }
  }
  function remove(root, key) {
    if (has(root, key)) { delete root[key]; mark(); }
  }
  function filter(root, path, isAd) {
    var parts = path.split(".");
    var key = parts.pop();
    var parent = parts.length ? get(root, parts.join(".")) : root;
    if (!has(parent, key) || !Array.isArray(parent[key])) return 0;
    var old = parent[key];
    var out = old.filter(function (item) { return !isAd(item); });
    if (out.length !== old.length) { parent[key] = out; mark(); }
    return old.length - out.length;
  }
  function each(root, path, fn) {
    var items = get(root, path);
    if (Array.isArray(items)) items.forEach(function (x) { if (object(x)) fn(x); });
  }

  // MTOP 的 data 有对象和 JSON 字符串两种包装；仅在实际修改后重新编码。
  function embedded(root, key, fn) {
    if (!has(root, key)) return;
    if (object(root[key])) { fn(root[key]); return; }
    if (typeof root[key] !== "string") return;
    var nested = parsePreservingIntegers(root[key]);
    if (!object(nested.value)) return;
    var previous = edits;
    fn(nested.value);
    if (edits !== previous) root[key] = nested.stringify(nested.value);
  }

  function queryValue(text, key) {
    if (typeof text !== "string") return "";
    var entries = text.split("&");
    for (var i = 0; i < entries.length; i++) {
      var at = entries[i].indexOf("=");
      if (at < 0) continue;
      var name = decodeURIComponent(entries[i].slice(0, at).replace(/\+/g, " "));
      if (name === key) return decodeURIComponent(entries[i].slice(at + 1).replace(/\+/g, " "));
    }
    return "";
  }

  // 只接受已经使用的广告类型值，不将任意包含 ad 的文本视为广告。
  function goofishAdType(value) {
    return typeof value === "string" && /^(?:ad|ads|advertisement|mamaad)$/i.test(value);
  }
  function goofishSectionAd(section) {
    return goofishAdType(get(section, "data.bizType")) || flag(get(section, "data.isAd"));

  function goofishSearchAd(item) {
    var main = get(item, "data.item.main");
    return flag(get(main, "exContent.isAliMaMaAD")) ||
      goofishAdType(get(main, "clickParam.args.item_type")) ||
      goofishAdType(get(main, "clickParam.args.biz_type"));
  }

  // JSON.parse 会舍入大于 2^53-1 的整数。先保护这些数值，再恢复原始数字文本。
  // 不匹配字符串内的数字；占位前缀在原响应中不得出现，防止误替换普通文本。
  function parsePreservingIntegers(text) {
    var prefix = "__ABV5_INTEGER_";
    var stringValues = [];
    text.replace(/"(?:\\.|[^"\\])*"/g, function (token) {
      stringValues.push(JSON.parse(token));
      return token;
    });
    // 检查解码后的字符串，防止 \u0049 等转义拼出占位符而误恢复成数字。
    while (text.indexOf(prefix) !== -1 || stringValues.some(function (value) { return value.indexOf(prefix) !== -1; })) prefix += "X";
    var numbers = [];
    var protectedText = text.replace(/"(?:\\.|[^"\\])*"|-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?/g, function (token) {
      if (token.charAt(0) === '"') return token;
      var n = Number(token);
      if (!isFinite(n)) throw new Error("non-finite numeric value");
      if (Math.abs(n) > 9007199254740991) {
        var index = numbers.push(token) - 1;
        return '"' + prefix + index + '__"';
      }
      return token;
    });
    return {
      value: JSON.parse(protectedText),
      stringify: function (value) {
        var result = JSON.stringify(value);
        if (!numbers.length) return result;
        return result.replace(new RegExp('"' + prefix + '(\\d+)__"', "g"), function (whole, i) {
          return numbers[Number(i)];
        });
      }
    };
  }

  function weiboAd(x) {

  // 只沿微博已知卡片容器遍历，不扫描正文和用户资料里的任意数组。
  function weiboContainers(root, depth) {
    if (!object(root) || depth > 8) return;
    ["statuses", "cards", "items", "card_group"].forEach(function (key) {
      filter(root, key, function (x) {
        if (weiboCard(x)) return true;
        if (!object(x)) return false;
        var groupSize = Array.isArray(x.card_group) ? x.card_group.length : 0;
        weiboContainers(x, depth + 1);
        // 已知 card_group 原本全部是广告，且父项不承载另一份正文时移除父壳。
        return groupSize > 0 && x.card_group.length === 0 &&
          !active(x.mblog) && !active(x.data) && !Array.isArray(x.statuses) &&
          !Array.isArray(x.items) && !Array.isArray(x.cards);
      });
    });
    if (object(root.data)) weiboContainers(root.data, depth + 1);
  }

  function ximaAd(x) {

  // 仅过滤基本、非加密的 VOD 播放列表，并要求广告区块中每个分片都满足证据。
  // KEY/MAP/BYTERANGE/直播/低延迟等格式原样保留，避免破坏解密和分片序列。
  function cleanPlaylist(text, kind, url) {
    if (typeof text !== "string" || !/^#EXTM3U(?:\r?\n|$)/.test(text) || !/#EXT-X-ENDLIST(?:\r?\n|$)/.test(text)) return text;
    if (/#EXT-X-(?:KEY|SESSION-KEY|MAP|BYTERANGE|DISCONTINUITY-SEQUENCE|PART|PRELOAD-HINT|SKIP|SERVER-CONTROL)(?::|\r?\n)/.test(text)) return text;
    if (/#EXT-X-(?:STREAM-INF|I-FRAME-STREAM-INF|MEDIA):/.test(text)) return text;
    var main = "";
    if (kind === "youku") {
      var idMatch = /[?&]vid=([^&#]+)/i.exec(url);
      if (!idMatch) return text;
      try { main = decodeURIComponent(idMatch[1]); } catch (_) { return text; }
    }
    function adURI(uri) {
      if (kind === "tencent") return /^https?:\/\/defaultts\.tc\.qq\.com\/(?:[^/?#]+\/)*svp_[^?#]*\?/i.test(uri) && /[?&]segmenttype=2(?:&|$)/i.test(uri);
      var m = /[?&]vid=([^&#]+)/i.exec(uri);
      if (!m || !/[?&]ccode=0902(?:&|$)/i.test(uri)) return false;
      try { return decodeURIComponent(m[1]) !== main; } catch (_) { return false; }
    }
    var eol = text.indexOf("\r\n") !== -1 ? "\r\n" : "\n";
    var lines = text.split(/\r?\n/);
    var groups = [], current = [];
    lines.forEach(function (line) {
      if (line === "#EXT-X-DISCONTINUITY") { groups.push(current); current = [line]; }
      else current.push(line);
    });
    groups.push(current);
    if (groups.some(function (group) {
      var media = group.filter(function (line) { return line.trim() && line.charAt(0) !== "#"; }).length;
      var durations = group.filter(function (line) { return /^#EXTINF:\d+(?:\.\d+)?(?:,|$)/.test(line); }).length;
      return media !== durations;
    })) return text;
    var removed = 0, normal = 0, output = [];
    var headerPrefixes = /^#(?:EXTM3U$|EXT-X-(?:VERSION|TARGETDURATION|MEDIA-SEQUENCE|PLAYLIST-TYPE|INDEPENDENT-SEGMENTS|ALLOW-CACHE|START)(?::|$))/;
    groups.forEach(function (group) {
      var uris = group.filter(function (line) { return line.trim() && line.charAt(0) !== "#"; });
      var isAd = uris.length > 0 && uris.every(adURI);
      if (isAd) {
        removed++;
        group.forEach(function (line) { if (headerPrefixes.test(line)) output.push(line); });
      } else {
        // 删除片头广告后保留原播放头部，但不留下开头的孤立不连续标记。
        if (removed > 0 && normal === 0 && group[0] === "#EXT-X-DISCONTINUITY") group = group.slice(1);
        output = output.concat(group.filter(function (line) { return line !== "#EXT-X-ENDLIST"; }));
        normal += uris.length;
      }
    });
    if (!removed || !normal) return text;
    while (output.length && output[output.length - 1] === "") output.pop();
    output.push("#EXT-X-ENDLIST");
    return output.join(eol) + (/\r?\n$/.test(text) ? eol : "");
  }

  function cainiaoList(root) {

      filter(data, "result.dataList", function (x) {

      });
    });
  }

  function processBody(root, host, path, query) {
    switch (label) {

      case "bcomic":
        if (host !== "manga.bilibili.com") break;
        if (/^\/twirp\/comic\.v\d+\.Comic\/ListFlash\/?$/.test(path)) {
          ["list", "ad_list", "show", "verifying_list"].forEach(function (key) {
            setExisting(root, "data." + key, []);
          });
          setExisting(root, "data.ad_loc", false);
        } else if (/^\/twirp\/comic\.v\d+\.Comic\/AppInit\/?$/.test(path)) {
          var flash = get(root, "data.flash");
          if (object(flash) && typeof flash.img === "string" && has(flash, "id")) {
            setExisting(flash, "id", 0);
            setExisting(flash, "show_limit", 0);
            ["img", "video", "back_img", "button"].forEach(function (key) { setExisting(flash, key, ""); });
          }
        }
        break;

      case "bili":
        if (host === "app.bilibili.com" && /^\/x\/v2\/feed\/index(?:\/story)?$/.test(path)) {
          filter(root, "data.items", biliAd);
        } else if (host === "api.bilibili.com" && path === "/pgc/page/bangumi") {
          filter(root, "result.modules", function (module) {
            var removed = filter(module, "items", function (x) { return object(x) && active(get(x, "source_content.ad_content")); });
            // 混合模块只删广告项；全是已确认广告的模块连同外壳一起移除。
            return removed > 0 && get(module, "items.length") === 0;
          });
        }
        break;
      case "zhihu":
        if ((host === "api.zhihu.com" || host === "web-render.zhihu.com") &&
          /^\/(?:topstory\/recommend|next-render|(?:v4\/)?questions\/\d+(?:\/answers)?|search\/recommend_query)$/.test(path)) {
          filter(root, "data", zhihuAd);
          filter(root, "recommend_queries.queries", zhihuAd);
          if (/\/questions\//.test(path)) { remove(root, "ad_info"); remove(root.data, "ad_info"); }
        } else if ((host === "api.zhihu.com" || host === "page-info.zhihu.com") &&
          /^\/(?:answers\/v2\/\d+|articles\/v\d+\/\d+)$/.test(path)) {
          remove(root, "ad_info");
          // third_business 也承载普通相关问题，保留整个业务对象。
          remove(root.third_business, "ad_info");
        }
        break;
      case "weibo":
        if (/^(?:m?api)\.weibo\.(?:cn|com)$/.test(host) &&
          /^\/\d+\/(?:statuses\/(?:[^/]*timeline|friends\/timeline|unread_friends\/timeline|container_detail|extend)|groups\/timeline|cardlist|searchall|page|0flowpage|search\/(?:container_timeline|container_discover|finder))$/.test(path)) {
          weiboContainers(root, 0);
          remove(root, "ad"); remove(root, "advertises");
        }
        break;
      case "coolapk":
        if (host === "api.coolapk.com" && /^\/v6\/(?:main\/(?:indexV8|dataList|init)|feed\/replyList|page\/dataList)$/.test(path)) {
          filter(root, "data", coolapkAd);
          if (path === "/v6/main/init") filter(root, "data", function (x) { return object(x) && (x.entityId === 945 || x.entityId === "945"); });
        } else if (host === "api.coolapk.com" && path === "/v6/feed/detail") {
          setExisting(root, "data.detailSponsorCard", []);
          if (object(get(root, "data.detailSponsorCard"))) remove(root.data, "detailSponsorCard");
          filter(root, "data.hotReplyRows", coolapkAd);
          filter(root, "data.topReplyRows", coolapkAd);
        }
        break;
      case "smzdm":
        if (host === "homepage-api.smzdm.com" && path === "/v3/home") {
          filter(root, "data.component", function (component) {
            if (!object(component) || (component.zz_type !== "banner" && component.zz_type !== "list")) return false;
            var removed = filter(component, "zz_content", smzdmAd);
            return removed > 0 && get(component, "zz_content.length") === 0;
          });
        } else if ((host === "haojia-api.smzdm.com" && path === "/home/list") ||
                   (host === "baike-api.smzdm.com" && path === "/home_v3/list") ||
                   (host === "s-api.smzdm.com" && path === "/sou/list_v10")) {
          filter(root, "data.rows", smzdmAd);
          filter(root, "data.banner_v2", smzdmAd);
        } else if (host === "app-api.smzdm.com" && path === "/util/loading") {
          each(root, "data", function (x) { setExisting(x, "is_show_ad", "0"); });
        }
        break;
      case "ximalaya":
        if (!/^(?:mobile|mobilehera|mobwsa|mobi)\.ximalaya\.com$/.test(host)) break;
        if (/^\/discovery-feed\/v\d+\/mix$/.test(path)) {
          filter(root, "body", ximaAd);
          filter(root, "header", ximaAd);
        } else if (/^\/discovery-category\/v\d+\/category$/.test(path)) {
          filter(root, "focusImages.data", ximaAd);
        } else if (path === "/focus-mobile/focusPic") {
          filter(root, "header", function (x) {
            var removedGroups = filter(x, "item.list", function (item) {
              var removed = filter(item, "data", ximaAd);
              return removed > 0 && get(item, "data.length") === 0;
            });
            return removedGroups > 0 && get(x, "item.list.length") === 0;
          });
        }
        break;
      case "jd":
        if (host !== "api.m.jd.com" || path !== "/client.action") break;
        var functionId = queryValue(query, "functionId");
        // 某些客户端将 functionId 放在 POST 表单或 JSON；只处理已知功能名。
        if (!functionId && typeof $request.body === "string") {
          if (/^\s*\{/.test($request.body)) {
            var jdRequest = parsePreservingIntegers($request.body).value;
            if (object(jdRequest) && typeof jdRequest.functionId === "string") functionId = jdRequest.functionId;
          } else functionId = queryValue($request.body, "functionId");
        }
        if (functionId === "start") {
          setExisting(root, "images", []);
          // 两字段独立处理；images 缺失或为空不应跳过次数开关。
          setExisting(root, "showTimesDaily", 0);
        } else if (functionId === "welcomeHome") {
          filter(root, "floorList", function (x) {
            return object(x) && ["bottomXview", "float", "photoCeiling", "tabBarAtmosphere"].indexOf(x.type) !== -1;
          });
        } else if (functionId === "deliverLayer" || functionId === "orderTrackBusiness") {
          remove(root, "bannerInfo");
          filter(root, "floors", function (x) { return object(x) && (x.mId === "banner" || x.mId === "jdDeliveryBanner"); });
        }
        break;
      case "meituan":
        if (host !== "wmapi.meituan.com" || !/^\/api\/v\d+\/loadInfo$/.test(path)) break;
        setExisting(root, "data.startpicture.ad", []);
        setExisting(root, "data.startpicture.mk", []);
        break;
      case "dmall":
        if (host !== "cmsapi.dmall.com" || path !== "/app/home/homepageStartUpPic") break;
        setExisting(root, "data.welcomePage", []);
        break;
      case "dida":
        if (!/^capis(?:-?\w*)?\.didapinche\.com$/.test(host) || path !== "/ad/cx/startup") break;
        setExisting(root, "startupPages", []);
        setExisting(root, "show_time", 0);
        setExisting(root, "full_screen", 0);
        break;
      case "xiaoai":
        if (host !== "hd.mina.mi.com" || path !== "/splashscreen/alert") break;
        if (Array.isArray(root.data)) setExisting(root, "data", []);
        break;
      case "family":
        if (host !== "fmapp.chinafamilymart.com.cn" || path !== "/api/app/market/start/ad") break;
        if ((root.code === "200" || root.code === 200) && object(root.data)) setExisting(root, "data", {});
        break;
      case "goofish":
        if (!/^(?:g-)?acs\.m\.(?:goofish|taobao)\.com$/.test(host)) break;
        embedded(root, "data", function (data) {
          if (/^\/gw\/mtop\.taobao\.(?:idlehome\.home\.nextfresh|idle\.local\.home)(?:\/|$)/.test(path)) {
            var count = Array.isArray(data.sections) ? data.sections.length : -1;
            var removed = filter(data, "sections", goofishSectionAd);
            // feedsCount 只有等于本批数组长度才同步；服务器总数/分页游标不猜测。
            if (removed && Number(data.feedsCount) === count) {
              setExisting(data, "feedsCount", typeof data.feedsCount === "string" ? String(data.sections.length) : data.sections.length);
            }
          } else if (/^\/gw\/mtop\.taobao\.idlemtopsearch\.search(?:\/|$)/.test(path)) {
            filter(data, "resultList", goofishSearchAd);
          } else if (/^\/gw\/mtop\.taobao\.idle\.item\.recommend(?:\.list)?(?:\/|$)/.test(path)) {
            filter(data, "cardList", function (x) { return goofishAdType(get(x, "cardData.bizType")); });
          } else if (/^\/gw\/mtop\.taobao\.idlecommerce\.splash\.async\.ads(?:\/|$)/.test(path)) {
            setExisting(data, "adMap", {});
          } else if (/^\/gw\/mtop\.taobao\.idlecommerce\.splash\.ads(?:\/|$)/.test(path)) {
            // 独立开屏广告 RPC：保留外层 api/ret/v，仅清空广告负载。
            Object.keys(data).forEach(function (key) { remove(data, key); });
          }
        });
        break;
      case "cainiao":
        if (!/^[a-z0-9.-]+\.cainiao\.com$/.test(host)) break;
        if (/^\/gw\/mtop\.cainiao\.nbpresentation\.protocol\.homepage\.get\.cn(?:\/|$)/.test(path)) {
          cainiaoList(root);
        } else if (/^\/gw\/mtop\.cainiao\.nbpresentation\.homepage\.merge\.get\.cn(?:\/|$)/.test(path)) {
          embedded(root, "data", function (data) {
            Object.keys(data).forEach(function (key) {
              if (/^mtop\.cainiao\.nbpresentation\.protocol\.homepage\.get\.cn@\d+$/.test(key)) {
                embedded(data, key, cainiaoList);
              }
            });
          });
        } else if (/^\/gw\/mtop\.cainiao\.adkeyword(?:\.|\/)/.test(path)) {
          embedded(root, "data", function (data) { setExisting(data, "result.adHotKeywords", []); });
        } else if (/^\/gw\/mtop\.cainiao\.guoguo\.nbnetflow\.ads\.show\.cn(?:\/|$)/.test(path)) {
          embedded(root, "data", function (data) {
            filter(data, "result", function (item) {
              var material = get(item, "materialContentMapper");
              return object(material) && (active(material.adItemDetail) ||
                (active(material.bgImg) && active(material.advRecGmtModifiedTime)) ||
                material.group_id === "common_header_banner" || material.group_id === "kuaishou_banner");
            });
          });
        }
        break;
      case "unicom":
        if (host !== "m.client.10010.com" || path !== "/mobileService/customer/getclientconfig.htm") break;
        var isString = typeof root.DTO === "string";
        var nested = isString ? parsePreservingIntegers(root.DTO) : null;
        var dto = isString ? nested.value : root.DTO;
        if (!object(dto)) break;
        ["adMaxTime", "startMaxTime"].forEach(function (key) {
          if (has(dto, key)) setExisting(dto, key, typeof dto[key] === "number" ? 0 : "0");
        });
        if (changed && isString) root.DTO = nested.stringify(dto);
        break;
      case "xhs":
        if (!/^(?:edith|www|rec|so|ci)\.xiaohongshu\.com$/.test(host)) break;
        if (/^\/api\/sns\/v6\/homefeed$/.test(path)) {
          filter(root, "data", function (x) { return object(x) && (object(x.ads_info) || active(x.ads_info)); });
        } else if (/^\/api\/sns\/v\d+\/search\/notes$/.test(path)) {
          filter(root, "data.items", function (x) {
            return object(x) && (object(x.ads_info) || active(x.ads_info) || x.model_type === "ad" || x.model_type === "ads");
          });
        } else if (/^\/api\/sns\/v\d+\/system_service\/splash_config$/.test(path)) {
          setExisting(root, "data.ads_groups", []);
        } else if (/^\/api\/sns\/v\d+\/system_service\/config$/.test(path)) {
          // 只删开屏配置，保留主题、商城及其他正常设置。
          remove(root.data, "splash");
        }
        break;
      case "tencent":
        if (!/^(?:s)?vv\.video\.qq\.com$/.test(host) || path !== "/getvinfo") break;
        each(root, "vl.vi", function (vi) {
          remove(vi, "ad");
          var playlist = get(vi, "ul.m3u8");
          var cleaned = cleanPlaylist(playlist, "tencent", $request.url);
          if (cleaned !== playlist) setExisting(vi, "ul.m3u8", cleaned);
        });
        break;
      case "youku":
        if (host !== "un-acs.youku.com" || !/^\/gw\/mtop\.youku\.play\.ups\.appinfo\.get\//.test(path)) break;
        var payload = get(root, "data.data");
        if (object(payload) && (object(payload.video) || Array.isArray(payload.stream))) {
          remove(payload, "ad"); remove(payload, "ykad");
        }
        break;
    }
  }

  try {
    if (typeof $request === "undefined") { $done({}); return; }
    var match = /^https?:\/\/([^/?#:]+)(?::\d+)?(\/[^?#]*)?(?:\?([^#]*))?/i.exec($request.url || "");
    if (!match) { $done({}); return; }
    if (typeof $response === "undefined") {
      if (label === "rail12306" && match[1] === "ad.12306.cn" && match[2] === "/ad/ser/getAdList") {
        var request = JSON.parse($request.body || "{}");
        if (!object(request) || typeof request.placementNo !== "string") { $done({}); return; }
        var answer = {code: "00", message: "无广告返回"};
        if (request.placementNo === "0007") answer = {materialsList: [{billMaterialsId: "6491", filePath: "ddgksf2013", creativeType: 1}], advertParam: {skipTime: 1}, code: "00"};
        else if (request.placementNo === "G0054") answer = {code: "00", materialsList: [{}]};
        $done({response: {status: 200, headers: {"Content-Type": "application/json; charset=utf-8", "Cache-Control": "no-store"}, body: JSON.stringify(answer)}});
      } else if (label === "tencent" && /^(?:s)?vv\.video\.qq\.com$/.test(match[1]) && match[2] === "/getvinfo" && typeof $request.body === "string") {
        var reqBody = $request.body.replace(/(^|&)(sppreviewtype|spsrt|spadseg)=[^&]*/gi, "$1$2=0");
        $done(reqBody !== $request.body ? {body: reqBody} : {});
      } else { $done({}); }
      return;
    }
    if (typeof $response.body !== "string" || !$response.body) { $done({}); return; }
    var status = $response.statusCode !== undefined ? $response.statusCode : $response.status;
    if (typeof status === "string") {
      var statusMatch = /^(?:HTTP\/\d(?:\.\d)?\s+)?(\d{3})(?:\s|$)/.exec(status);
      if (statusMatch) status = Number(statusMatch[1]);
    }
    if (typeof status === "number" && (status < 200 || status >= 300)) { $done({}); return; }
    if (label === "youku" && match[1] === "pl-ali.youku.com" && match[2] === "/playlist/m3u8") {
      var hls = cleanPlaylist($response.body, "youku", $request.url);
      $done(hls !== $response.body ? {body: hls} : {});
      return;
    }
    var parsed = parsePreservingIntegers($response.body);
    if (!object(parsed.value)) { $done({}); return; }
    var responsePath = match[2] || "/";
    // 只规范普通 JSON 接口的单个尾斜杠；腾讯/优酷播放分支保留原判断。
    if (label !== "tencent" && label !== "youku" && responsePath.length > 1 && !/\/$/.test(responsePath.slice(0, -1))) responsePath = responsePath.replace(/\/$/, "");
    processBody(parsed.value, match[1].toLowerCase(), responsePath, match[3] || "");
    if (changed) {
      var body = parsed.stringify(parsed.value);
      console.log("[AdBlock V5.1] " + label + ": " + edits + " changes");
      $done({body: body});
    } else {
      if (debug) console.log("[AdBlock V5.1] " + label + ": unchanged");
      $done({});
    }
  } catch (e) {
    // 不输出响应、URL 参数或异常中的输入片段。
    console.log("[AdBlock V5] " + label + ": passthrough (" + (e && e.name || "Error") + ")");
    $done({});
  }
})();
