  function bytes(value) {
    if (value instanceof Uint8Array) return value;
    if (typeof ArrayBuffer !== "undefined" && value instanceof ArrayBuffer) return new Uint8Array(value);

  function readVarint(b, pos, end) {
    let value = 0;
    let factor = 1;
    for (let i = 0; i < 10; i++) {
      if (pos >= end) throw new Error("truncated-varint");
      const c = b[pos++];
      if (i === 9 && c > 1) throw new Error("overflow-varint");
      value += (c & 127) * factor;
      if (!(c & 128)) return { value: Number.isSafeInteger(value) ? value : null, next: pos };
      factor *= 128;
    }
    throw new Error("overlong-varint");
  }

  function parse(b) {

      p = tag.next;
      if (tag.value === null || tag.value < 8 || tag.value > 4294967295) throw new Error("bad-tag");
      const n = Math.floor(tag.value / 8);
      const w = tag.value % 8;
      const tagEnd = p;
      let valueStart = p;
      let value = null;
      if (w === 0) {
        const v = readVarint(b, p, b.length);
        value = v.value;
        p = v.next;
      } else if (w === 1 || w === 5) {
        p += w === 1 ? 8 : 4;
      } else if (w === 2) {
        const len = readVarint(b, p, b.length);
        if (len.value === null || len.value > MAX_BODY) throw new Error("bad-length");
        valueStart = len.next;
        p = valueStart + len.value;
      } else {
        throw new Error("unsupported-wire");
      }
      if (p > b.length) throw new Error("field-overrun");
      out.push({ n, w, value, raw: b.subarray(start, p), tag: b.subarray(start, tagEnd), data: b.subarray(valueStart, p) });
    }
    return out;
  }

  function textEquals(b, s) {

  }

  function contains(b, s) {
    outer: for (let i = 0; i <= b.length - s.length; i++) {

      return true;
    }
    return false;
  }

  function encodeVarint(v) {
    const out = [];
    do {
      const c = v % 128;
      v = Math.floor(v / 128);
      out.push(c | (v ? 128 : 0));
    } while (v);
    return new Uint8Array(out);
  }

  function concat(parts) {

    for (const part of parts) { out.set(part, p); p += part.length; }

  }

/* Codec adapter written for this task. Embedded pako vendor code is unmodified.
 * Input buffers are read only. Only one complete RFC1952 gzip member is accepted.
 * Output limit is enforced while decoding; maxOutput must be 0..32 MiB.
 * A chunk may temporarily exceed the requested limit by at most 16 KiB before
 * the exception; no oversized result is returned. All parse errors throw.
 * Header size limit: 64 KiB. Compressed input size limit: 64 MiB.
 */

var __SurgeCrcTable = (function () {
  var table = new Uint32Array(256);
  for (var i = 0; i < 256; i++) {
    var crc = i;
    for (var bit = 0; bit < 8; bit++) crc = (crc >>> 1) ^ ((crc & 1) ? 0xedb88320 : 0);
    table[i] = crc >>> 0;
  }
  return table;
})();

function codecCrc32(bytes, end) {
  var crc = 0xffffffff;
  if (end === undefined) end = bytes.length;
  for (var i = 0; i < end; i++) crc = (crc >>> 8) ^ __SurgeCrcTable[(crc ^ bytes[i]) & 255];
  return (crc ^ 0xffffffff) >>> 0;
}

function codecReadLE32(bytes, offset) {
  return (bytes[offset] + bytes[offset + 1] * 256 + bytes[offset + 2] * 65536 + bytes[offset + 3] * 16777216) >>> 0;
}

function codecWriteLE32(bytes, offset, value) {
  bytes[offset] = value & 255; bytes[offset + 1] = (value >>> 8) & 255;
  bytes[offset + 2] = (value >>> 16) & 255; bytes[offset + 3] = value >>> 24;
}

function inflateGzip(input, maxOutput) {
  if (!(input instanceof Uint8Array)) throw new TypeError("gzip input must be Uint8Array");
  if (typeof maxOutput !== "number" || maxOutput !== Math.floor(maxOutput) ||
      maxOutput < 0 || maxOutput > 33554432) throw new RangeError("invalid gzip output limit");
  if (input.length < 18 || input.length > 67108864) throw new RangeError("invalid gzip input length");
  if (input[0] !== 31 || input[1] !== 139 || input[2] !== 8 || (input[3] & 224))
    throw new Error("invalid RFC1952 header");
  var flags = input[3], position = 10, headerLimit = Math.min(65536, input.length - 8);
  if (flags & 4) {
    if (position + 2 > headerLimit) throw new Error("truncated gzip extra length");
    position += 2 + input[position] + input[position + 1] * 256;
    if (position > headerLimit) throw new Error("invalid gzip extra length");
  }
  for (var field = 8; field <= 16; field *= 2) {
    if (flags & field) {
      while (position < headerLimit && input[position] !== 0) position++;
      if (position >= headerLimit) throw new Error("unterminated gzip header field");
      position++;
    }
  }
  if (flags & 2) {
    if (position + 2 > headerLimit) throw new Error("truncated gzip header CRC");
    if ((codecCrc32(input, position) & 65535) !== input[position] + input[position + 1] * 256)
      throw new Error("gzip header CRC mismatch");
    position += 2;
  }
  if (position >= input.length - 8) throw new Error("missing gzip deflate payload");
  var chunks = [], total = 0;
  var decoder = new __SurgePakoInflate.Inflate({windowBits: 31, chunkSize: 16384});
  decoder.onData = function (chunk) {
    if (chunk.length > maxOutput - total) throw new RangeError("gzip output limit exceeded");
    chunks.push(chunk); total += chunk.length;
  };
  // false is critical: pako 1.0.x push(..., true) may finalize a truncated
  // stream without seeing DEFLATE end / gzip trailer. With false, ended
  // becomes true on successful Z_STREAM_END only (or an error checked below).
  decoder.onEnd = function (status) { this.err = status; this.msg = this.strm.msg; };
  var ok = decoder.push(input, false);
  if (!ok || decoder.err) throw new Error("invalid gzip stream: " + (decoder.msg || decoder.err));
  if (!decoder.ended) throw new Error("truncated gzip stream");
  if (decoder.strm.avail_in !== 0 || decoder.strm.next_in !== input.length)
    throw new Error("trailing bytes after gzip member");

  if (codecReadLE32(input, input.length - 4) !== total) throw new Error("gzip ISIZE mismatch");
  if (codecCrc32(output) !== codecReadLE32(input, input.length - 8)) throw new Error("gzip CRC-32 mismatch");
  return output;
}

function encodeGzipStored(input) {
  if (!(input instanceof Uint8Array)) throw new TypeError("gzip input must be Uint8Array");
  if (input.length > 33554432) throw new RangeError("gzip input exceeds 32 MiB");
  var blocks = Math.max(1, Math.ceil(input.length / 65535));
  var output = new Uint8Array(10 + 5 * blocks + input.length + 8);
  // CM=DEFLATE, FLG=0, MTIME=0, XFL=0, OS=255. Header fields are deterministic.
  output[0] = 31; output[1] = 139; output[2] = 8; output[9] = 255;
  var offset = 10, read = 0;
  for (var i = 0; i < blocks; i++) {
    var length = Math.min(65535, input.length - read), complement = length ^ 65535;
    output[offset++] = i === blocks - 1 ? 1 : 0; // BFINAL then BTYPE=00, pad to byte.
    output[offset++] = length & 255; output[offset++] = length >>> 8;
    output[offset++] = complement & 255; output[offset++] = complement >>> 8;
    output.set(input.subarray(read, read + length), offset);
    offset += length; read += length;
  }
  codecWriteLE32(output, offset, codecCrc32(input));
  codecWriteLE32(output, offset + 4, input.length);
  return output;
}
