 * Surge http-response, requires-body=true, binary-body-mode=true.
 * Evidence: 2026-09-08-100604.har, entries 285 and 363 (one-based).
 * Scope: existing channel$100101 homepage plus the exact captured reward-ad pendant.
 * Added evidence: 2026-09-09-003815.har entry 823 (one-based).
 * Only display child #2/#2 is removed; state/reward/time siblings stay byte-identical.
 * New pendant branch is an offline-verified candidate; phone UI effect needs confirmation.
 * Playback stays with the existing TenVideo-MVL-AdFilter-Test.js?v=22.
 * No request changes, VIP flags, token storage, HTTP calls or CDN blocking.
 * Parse real protobuf boundaries, remove confirmed repeated ad items,
 * rebuild enclosing lengths and the tRPC frame length. Never rename tags.
 * Unknown/malformed/compressed-inner frames pass through unchanged.

(function () {
  "use strict";
  const VERSION = "TencentVideo PageClean V1.1.0 RC1";
  const PREFIX = "type.googleapis.com/com.tencent.qqlive.protocol.pb.";
  const MAX_BODY = 2097152;
  const MAX_FIELDS = 20000;
  let parsedFields = 0;

  function finish(value) { $done(value || {}); }

  function header(headers, name) {
    name = name.toLowerCase();
    if (Array.isArray(headers)) {
      const matches = headers.filter(h => String(h.field || "").toLowerCase() === name);
      return matches.length === 1 ? String(matches[0].value) : "";
    }
    const keys = Object.keys(headers || {}).filter(k => k.toLowerCase() === name);
    return keys.length === 1 ? String(headers[keys[0]]) : "";
  }

  function bytes(value) {
    if (value instanceof Uint8Array) return value;
    if (typeof ArrayBuffer !== "undefined" && value instanceof ArrayBuffer) return new Uint8Array(value);
    return null;
  }

  function readVarint(b, pos, end) {
    let value = 0;
    let factor = 1;
    for (let i = 0; i < 10; i++) {
      if (pos >= end) throw new Error("truncated-varint");
      const c = b[pos++];
      if (i === 9 && c > 1) throw new Error("overflow-varint");
      value += (c & 127) * factor;
      if (!(c & 128)) return { value: Number.isSafeInteger(value) ? value : null, next: pos };
      factor *= 128;
    }
    throw new Error("overlong-varint");
  }

  function parse(b) {
    const out = [];
    for (let p = 0; p < b.length;) {
      if (++parsedFields > MAX_FIELDS) throw new Error("field-budget");
      const start = p;
      const tag = readVarint(b, p, b.length);
      p = tag.next;
      if (tag.value === null || tag.value < 8 || tag.value > 4294967295) throw new Error("bad-tag");
      const n = Math.floor(tag.value / 8);
      const w = tag.value % 8;
      const tagEnd = p;
      let valueStart = p;
      let value = null;
      if (w === 0) {
        const v = readVarint(b, p, b.length);
        value = v.value;
        p = v.next;
      } else if (w === 1 || w === 5) {
        p += w === 1 ? 8 : 4;
      } else if (w === 2) {
        const len = readVarint(b, p, b.length);
        if (len.value === null || len.value > MAX_BODY) throw new Error("bad-length");
        valueStart = len.next;
        p = valueStart + len.value;
      } else {
        throw new Error("unsupported-wire");
      }
      if (p > b.length) throw new Error("field-overrun");
      out.push({ n, w, value, raw: b.subarray(start, p), tag: b.subarray(start, tagEnd), data: b.subarray(valueStart, p) });
    }
    return out;
  }

  function one(fs, n, w) {
    const matches = fs.filter(f => f.n === n);
    return matches.length === 1 && matches[0].w === w ? matches[0] : null;
  }

  function numeric(fs, n, defaultValue) {
    const matches = fs.filter(f => f.n === n);
    if (!matches.length) return defaultValue;
    if (matches.length !== 1 || matches[0].w !== 0 || matches[0].value === null) return null;
    return matches[0].value;
  }

  function textEquals(b, s) {
    if (b.length !== s.length) return false;
    for (let i = 0; i < s.length; i++) if (b[i] !== s.charCodeAt(i)) return false;
    return true;
  }

  function contains(b, s) {
    outer: for (let i = 0; i <= b.length - s.length; i++) {
      for (let j = 0; j < s.length; j++) if (b[i + j] !== s.charCodeAt(j)) continue outer;
      return true;
    }
    return false;
  }

  function encodeVarint(v) {
    const out = [];
    do {
      const c = v % 128;
      v = Math.floor(v / 128);
      out.push(c | (v ? 128 : 0));
    } while (v);
    return new Uint8Array(out);
  }

  function concat(parts) {
    const length = parts.reduce((n, p) => n + p.length, 0);
    if (length > MAX_BODY) throw new Error("output-limit");
    const out = new Uint8Array(length);
    let p = 0;
    for (const part of parts) { out.set(part, p); p += part.length; }
    return out;
  }

  function wrap(field, payload) {
    return concat([field.tag, encodeVarint(payload.length), payload]);
  }

  function rebuild(fs, replacements) {
    return concat(fs.map(f => replacements.has(f) ? replacements.get(f) : f.raw));
  }

  function anyPayload(b, name) {
    const fs = parse(b);
    if (fs.length !== 2) return null;
    const type = one(fs, 1, 2), payload = one(fs, 2, 2);
    return type && payload && textEquals(type.data, PREFIX + name) ? payload.data : null;
  }

  function extensionPayload(field, kind, name) {
    const fs = parse(field.data);
    if (fs.length !== 2 || numeric(fs, 1, -1) !== kind) return null;
    const any = one(fs, 2, 2);
    return any ? anyPayload(any.data, name) : null;
  }

  // A real Block with an exact AdFeedInfo Any and both tracking markers.
  function confirmedAdBlock(b, mode) {
    const fs = parse(b);
    if (numeric(fs, 1, -1) !== 16 || numeric(fs, 2, -1) !== mode) return false;
    const any = one(fs, 4, 2);
    if (!any) return false;
    const ad = anyPayload(any.data, "AdFeedInfo");
    if (!ad || !contains(ad, "gdt_stats.fcg") || !contains(ad, "ad_request_id")) return false;
    const adFields = parse(ad);
    const poster = one(adFields, 3, 2);
    if (!poster) return false;
    if (mode === 1) return anyPayload(poster.data, "AdFocusPoster") !== null;
    return anyPayload(poster.data, "AdFeedImagePoster") !== null ||
      anyPayload(poster.data, "AdFeedVideoPoster") !== null;
  }

  // Captured feed module: a sole field #10 wraps Module -> one ad Block.
  // Do not remove a Module containing several blocks or unknown sibling fields.
  function isFeedAd(field) {
    const fs = parse(field.data);
    if (fs.length !== 1) return false;
    const ext = one(fs, 10, 2);
    if (!ext) return false;
    const module = extensionPayload(ext, 1, "Module");
    if (!module) return false;
    const moduleFields = parse(module);
    if (moduleFields.some(f => ![1, 2, 3, 6].includes(f.n) || f.w !== 2)) return false;
    const blockList = one(moduleFields, 2, 2);
    if (!blockList) return false;
    const listFields = parse(blockList.data);
    const list = one(listFields, 5, 2);
    if (!list || listFields.some(f => ![4, 5, 8].includes(f.n) || f.w !== 2)) return false;
    const items = parse(list.data);
    return items.length === 1 && items[0].n === 1 && items[0].w === 2 && confirmedAdBlock(items[0].data, 3);
  }

  function isFocusAd(field) {
    const fs = parse(field.data);
    if (fs.some(f => ![1, 2, 10].includes(f.n))) return false;
    if (!one(fs, 1, 2) || !one(fs, 2, 2)) return false;
    const ext = one(fs, 10, 2);
    if (!ext) return false;
    const block = extensionPayload(ext, 2, "Block");
    return block !== null && confirmedAdBlock(block, 1);
  }

  // Only the captured carousel map key "0". Keep every normal carousel item.
  function cleanCarousel(moduleField, counts) {
    const fs = parse(moduleField.data);
    const changes = new Map();
    for (const f of fs) {
      if (f.n !== 4 || f.w !== 2) continue;
      const mp = parse(f.data);
      if (mp.length !== 2) continue;
      const key = one(mp, 1, 2), value = one(mp, 2, 2);
      if (!key || !value || !textEquals(key.data, "0")) continue;
      const items = parse(value.data);
      if (!items.length || items.some(x => x.n !== 1 || x.w !== 2)) continue;
      const keep = items.filter(x => !isFocusAd(x));
      if (keep.length === items.length || keep.length === 0) continue;
      counts.focus += items.length - keep.length;
      const list = concat(keep.map(x => x.raw));
      changes.set(f, wrap(f, rebuild(mp, new Map([[value, wrap(value, list)]]))));
    }
    return changes.size ? wrap(moduleField, rebuild(fs, changes)) : moduleField.raw;
  }

  function extensionName(field) {
    const fs = parse(field.data);
    if (fs.length !== 2) return null;
    const any = one(fs, 2, 2);
    if (!any) return null;
    const kind = numeric(fs, 1, -1);
    if (kind === 5) {
      const ad = anyPayload(any.data, "AdResponseInfo");
      if (ad && contains(ad, "ad_request_id") && contains(ad, "feeds_ad_style")) return "adResponse";
    }
    if (kind === 7) {
      const ad = anyPayload(any.data, "InnerAdPromotionEventList");
      if (ad && contains(ad, "s.iwan.qq.com/opengame") &&
          contains(ad, PREFIX + "InnerAdPullRefreshEventList")) return "innerPromotion";
    }
    return null;
  }

  function read32(b, p) {
    return b[p] * 16777216 + b[p + 1] * 65536 + b[p + 2] * 256 + b[p + 3];
  }

  function cleanFrame(b) {
    if (b.length < 16 || b.length > MAX_BODY || b[0] !== 9 || b[1] !== 48 ||
        b[2] !== 0 || b[3] !== 0 || read32(b, 4) !== b.length) return null;
    const offset = 16 + b[8] * 256 + b[9];
    if (offset < 16 || offset >= b.length) return null;
    const meta = parse(b.subarray(16, offset));
    // Known plain protobuf success frames only; no attachments or inner compression.
    if ([1, 2, 4, 5, 9, 10, 12].some(n => numeric(meta, n, 0) !== 0)) return null;
    const fs = parse(b.subarray(offset));
    const extensions = one(fs, 8, 2);
    if (!extensions) return null;
    const extItems = parse(extensions.data);
    if (extItems.some(f => f.n !== 1 || f.w !== 2)) return null;
    const channelConfig = extItems.some(f => {
      const x = parse(f.data), any = one(x, 2, 2);
      return any && anyPayload(any.data, "ChannelPageConfig") !== null;
    });
    if (!channelConfig) return null;

    const counts = { feed: 0, focus: 0, adResponse: 0, innerPromotion: 0 };
    const changes = new Map();
    let normalMaps = 0;
    for (const f of fs) {
      if (f.n !== 1 || f.w !== 2) continue;
      const mp = parse(f.data);
      if (mp.length !== 2) continue;
      const key = one(mp, 1, 2), value = one(mp, 2, 2);
      if (!key || !value || !textEquals(key.data, "normal")) continue;
      if (++normalMaps > 1) return null;
      const modules = parse(value.data);
      if (modules.some(m => m.n !== 1 || m.w !== 2)) return null;
      const parts = [];
      const before = counts.feed + counts.focus;
      for (const module of modules) {
        if (isFeedAd(module)) { counts.feed++; continue; }
        parts.push(cleanCarousel(module, counts));
      }
      // Do not create a wholly empty homepage from an unfamiliar all-ad response.
      if (!parts.length) return null;
      if (counts.feed + counts.focus > before) {
        changes.set(f, wrap(f, rebuild(mp, new Map([[value, wrap(value, concat(parts))]]))));
      }
    }
    if (normalMaps !== 1) return null;

    const keptExtensions = [];
    for (const ext of extItems) {
      const kind = extensionName(ext);
      if (kind) counts[kind]++;
      else keptExtensions.push(ext.raw);
    }
    if (counts.adResponse + counts.innerPromotion) changes.set(extensions, wrap(extensions, concat(keptExtensions)));
    if (!changes.size) return null;
    const output = concat([b.subarray(0, offset), rebuild(fs, changes)]);
    const len = output.length;
    output[4] = Math.floor(len / 16777216) & 255;
    output[5] = Math.floor(len / 65536) & 255;
    output[6] = Math.floor(len / 256) & 255;
    output[7] = len & 255;
    return { body: output, counts };
  }

  // RC4: only the captured reward-ad pendant display schema. No state changes.
  function cleanRewardPromotionPayload(fs, h) {
    const { parse, one, numeric, textEquals, contains, concat, wrap } = h;
    function shape(fields, schema, repeated = []) {
      if (fields.some(f => !Object.prototype.hasOwnProperty.call(schema, f.n) || schema[f.n] !== f.w)) return false;
      return Object.keys(schema).every(key => {
        const n = Number(key), count = fields.filter(f => f.n === n).length;
        return repeated.includes(n) ? count > 0 : count === 1;
      });
    }
    function fixedBytes(b, expected) {
      return b.length === expected.length && expected.every((v, i) => b[i] === v);
    }
    function event(fields, kind, key) {
      return fields.some(f => {
        if (f.n !== 4 || f.w !== 2) return false;
        const e = parse(f.data);
        if (!shape(e, { 1: 0, 2: 2 }) || numeric(e, 1, -1) !== kind) return false;
        const list = parse(one(e, 2, 2).data);
        if (!shape(list, { 1: 2 })) return false;
        const item = parse(one(list, 1, 2).data);
        return shape(item, { 1: 2, 2: 0 }) &&
          textEquals(one(item, 1, 2).data, key) && numeric(item, 2, -1) > 0;
      });
    }
    if (!shape(fs, { 2: 2 })) return null;
    const envelope = one(fs, 2, 2), fields = parse(envelope.data);
    if (!shape(fields, { 1: 2, 2: 2, 3: 0, 4: 2, 6: 0, 7: 2, 10: 2, 11: 2 }, [4])) return null;
    if (!event(fields, 4, "pendant_frequency_feed_back")) return null;
    const display = one(fields, 2, 2), ui = parse(display.data);
    if (!shape(ui, { 1: 2, 2: 2, 3: 2, 4: 0, 5: 0, 6: 2, 7: 2, 8: 2, 11: 2, 17: 2, 18: 2, 22: 0, 23: 2, 25: 0 }, [17])) return null;
    if (!contains(one(ui, 1, 2).data, "ad_request_id")) return null;
    // The captured headline starts with çå¹¿åï¼åè´¹ç and includes VIP.
    // Guard the fixed meaning without hard-coding the offered minute count.
    const headline = one(ui, 3, 2).data;
    const intro = [0xe7, 0x9c, 0x8b, 0xe5, 0xb9, 0xbf, 0xe5, 0x91, 0x8a, 0xef, 0xbc, 0x8c, 0xe5, 0x85, 0x8d, 0xe8, 0xb4, 0xb9, 0xe7, 0x9c, 0x8b];
    if (!fixedBytes(headline.subarray(0, intro.length), intro) || !contains(headline, "VIP")) return null;
    // UTF-8 for the captured fixed label: åè´¹ç.
    if (!fixedBytes(one(ui, 8, 2).data, [0xe5, 0x85, 0x8d, 0xe8, 0xb4, 0xb9, 0xe7, 0x9c, 0x8b])) return null;
    const maps = ui.filter(f => f.n === 17), wanted = { link_type: "1", ad_pr_id: "203", ad_pr_type: "1" };
    if (maps.length !== 3) return null;
    const seen = new Set();
    for (const f of maps) {
      const map = parse(f.data);
      if (!shape(map, { 1: 2, 2: 2 })) return null;
      const key = Object.keys(wanted).find(k => textEquals(one(map, 1, 2).data, k));
      if (!key || seen.has(key) || !textEquals(one(map, 2, 2).data, wanted[key])) return null;
      seen.add(key);
    }
    const popup = parse(one(ui, 18, 2).data);
    if (!shape(popup, { 2: 2, 3: 0, 4: 2 }, [4])) return null;
    const popupText = parse(one(popup, 2, 2).data);
    if (!shape(popupText, { 1: 2, 2: 2, 3: 2, 4: 2 })) return null;
    if (!event(popup, 2, "pendant_animation_frequency_exposure")) return null;
    const action = parse(one(ui, 23, 2).data);
    if (!shape(action, { 1: 0, 2: 2 }) || numeric(action, 1, -1) !== 2) return null;
    const actionPayload = parse(one(action, 2, 2).data);
    if (!shape(actionPayload, { 1: 0, 2: 2 }) || numeric(actionPayload, 1, -1) !== 5) return null;
    const link = parse(one(actionPayload, 2, 2).data);
    if (!shape(link, { 1: 2 }) || !contains(one(link, 1, 2).data, "ad_request_id")) return null;
    return concat([wrap(envelope, concat(fields.filter(f => f !== display).map(f => f.raw)))]);
  }
  

  function cleanRewardFrame(b) {
    if (b.length < 256 || b.length > 65536 || b[0] !== 9 || b[1] !== 48 ||
        b[2] !== 0 || b[3] !== 0 || read32(b, 4) !== b.length) return null;
    const offset = 16 + b[8] * 256 + b[9];
    if (offset < 16 || offset >= b.length) return null;
    const meta = parse(b.subarray(16, offset));
    if ([1, 2, 4, 5, 9, 10, 12].some(n => numeric(meta, n, 0) !== 0)) return null;
    const payload = cleanRewardPromotionPayload(parse(b.subarray(offset)),
      { parse, one, numeric, textEquals, contains, concat, wrap });
    if (!payload) return null;
    const output = concat([b.subarray(0, offset), payload]);
    const len = output.length;
    output[4] = Math.floor(len / 16777216) & 255;
    output[5] = Math.floor(len / 65536) & 255;
    output[6] = Math.floor(len / 256) & 255;
    output[7] = len & 255;
    return { body: output, rewardPendant: 1 };
  }

  function hasHeader(headers, name) {
    name = name.toLowerCase();
    return Array.isArray(headers)
      ? headers.some(h => String(h.field || "").toLowerCase() === name)
      : Object.keys(headers || {}).some(key => key.toLowerCase() === name);
  }

  try {
    if (typeof $request === "undefined" || typeof $response === "undefined" ||
        !/^https:\/\/i\.video\.qq\.com\/$/i.test($request.url || "") ||
        $request.method !== "POST" || Number($response.status) !== 200) {
      finish(); return;
    }
    const serviceTag = header($request.headers, "servicetag");
    const rewardCandidate = !hasHeader($request.headers, "servicetag");
    if (serviceTag !== "channel$100101" && !rewardCandidate) { finish(); return; }
    const b = bytes($response.body);
    if (!b || !/^application\/octet-stream(?:\s*;|$)/i.test(header($response.headers, "content-type"))) {
      finish(); return;
    }
    const result = rewardCandidate ? cleanRewardFrame(b) : cleanFrame(b);
    if (!result) { finish(); return; }
    if (result.rewardPendant) {
      console.log(VERSION + ": rewardPendant=1");
      finish({ body: result.body }); return;
    }
    console.log(VERSION + ": feed=" + result.counts.feed + ", focus=" + result.counts.focus +
      ", adResponse=" + result.counts.adResponse + ", innerPromotion=" + result.counts.innerPromotion);
    finish({ body: result.body });
  } catch (e) {
    // Never return a partly edited body, an empty response or a replacement status.
    console.log(VERSION + ": passthrough (unrecognized frame)");
    finish();
  }
})();
