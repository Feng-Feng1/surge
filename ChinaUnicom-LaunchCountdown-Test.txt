/*
 * 中国联通开屏倒计时净化 Test V1
 *
 * 2026-09-04 HAR 直接证据：
 * - accountListData.htm 已返回“黑名单用户不展示开屏广告”；
 * - getclientconfig.htm 仍下发 adMaxTime=5、startMaxTime=3；
 * - 因而广告已经没有，剩余等待由客户端启动配置保留。
 *
 * 本脚本只修改两个 HAR 直接证明的倒计时字段，其他联通配置原样保留。
 * 文件使用 .txt 后缀，但内容仍是 Surge JavaScript，可直接作为 script-path。
 */

(function () {
  const url = ($request && $request.url) || "";

  if (
    !/^https?:\/\/m\.client\.10010\.com\/mobileService\/customer\/getclientconfig\.htm(?:\?|$)/i.test(
      url
    )
  ) {
    $done({});
    return;
  }

  try {
    const body = $response && $response.body;

    if (typeof body !== "string" || body.length === 0) {
      $done({});
      return;
    }

    const outer = JSON.parse(body);
    const dtoWasString = typeof outer.DTO === "string";
    const dto = dtoWasString ? JSON.parse(outer.DTO) : outer.DTO;

    if (!dto || typeof dto !== "object") {
      $done({});
      return;
    }

    const changed = [];

    function setField(name, value) {
      if (!Object.prototype.hasOwnProperty.call(dto, name)) return;

      if (String(dto[name]) !== String(value)) {
        changed.push(name + ":" + String(dto[name]) + "->" + String(value));
        dto[name] = value;
      }
    }

    // HAR 中直接出现的开屏最长显示时间和启动最长等待时间。
    setField("adMaxTime", "0");
    setField("startMaxTime", "0");

    if (changed.length === 0) {
      $done({});
      return;
    }

    outer.DTO = dtoWasString ? JSON.stringify(dto) : dto;

    console.log(
      "ChinaUnicom launch countdown neutralized: " + changed.join(", ")
    );

    $done({ body: JSON.stringify(outer) });
  } catch (e) {
    console.log("ChinaUnicom launch countdown error: " + e);
    $done({});
  }
})();
