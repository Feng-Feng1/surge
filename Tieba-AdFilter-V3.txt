// Parse only actual protobuf boundaries. Untouched fields remain byte-identical.
function read32(bytes, start) {
  let n = 0, pos = start;
  for (let shift = 0; shift <= 28 && pos < bytes.length; shift += 7) {
    const b = bytes[pos++];
    if (shift === 28 && b > 15) throw Error("uint32 overflow");
    n += (b & 127) * Math.pow(2, shift);
    if (!(b & 128)) return { n, pos };
  }
  throw Error("truncated uint32");
}
function wireFields(bytes) {
  const list = [];
  let pos = 0;
  while (pos < bytes.length) {
    const start = pos, tag = read32(bytes, pos), no = Math.floor(tag.n / 8), wire = tag.n & 7;
    if (!no || no > 536870911) throw Error("bad field");
    pos = tag.pos;
    let begin = pos, value = null;
    if (wire === 0) {
      let ended = false;
      for (let i = 0; i < 10 && pos < bytes.length; i++) {
        const b = bytes[pos++];
        if (i === 9 && b > 1) throw Error("uint64 overflow");
        if (!(b & 128)) { ended = true; break; }
      }
      if (!ended) throw Error("truncated uint64");
      if (pos === begin + 1) value = bytes[begin];
    } else if (wire === 1) pos += 8;
    else if (wire === 2) { const len = read32(bytes, pos); begin = len.pos; pos = begin + len.n; }
    else if (wire === 5) pos += 4;
    else throw Error("unsupported group wire type");
    if (pos > bytes.length) throw Error("truncated field");
    list.push({ no, wire, value, raw: bytes.subarray(start, pos), tag: bytes.subarray(start, tag.pos), data: bytes.subarray(begin, pos) });
  }
  return list;
}
function var32(n) {
  const out = [];
  do { const b = n % 128; n = Math.floor(n / 128); out.push(n ? b | 128 : b); } while (n);
  return new Uint8Array(out);
}
function concat(parts) {
  const out = new Uint8Array(parts.reduce((n, p) => n + p.length, 0));
  let pos = 0;
  for (const p of parts) { out.set(p, pos); pos += p.length; }
  return out;
}
function rebuilt(field, data) { return concat([field.tag, var32(data.length), data]); }
function ascii(bytes) { let out = ""; for (const b of bytes) { if (b > 127) return ""; out += String.fromCharCode(b); } return out; }
function contains(bytes, text) {
  outer: for (let i = 0; i <= bytes.length - text.length; i++) {
    for (let j = 0; j < text.length; j++) if (bytes[i + j] !== text.charCodeAt(j)) continue outer;
    return true;
  }
  return false;
}
// null means no change; an empty Uint8Array means the container became empty.
function rewriteMessage(bytes, transform) {
  let changed = false;
  const parts = [];
  for (const f of wireFields(bytes)) {
    const replacement = transform(f);
    if (replacement === null) parts.push(f.raw);
    else { changed = true; if (replacement.length) parts.push(replacement); }
  }
  return changed ? concat(parts) : null;
}
function nested(field, transform, removeEmpty) {
  const result = transform(field.data);
  return result === null ? null : (removeEmpty && result.length === 0 ? new Uint8Array(0) : rebuilt(field, result));
}
